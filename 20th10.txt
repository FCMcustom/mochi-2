  <script>
    /* ================= 1. QUẢN LÝ CHUYỂN MÀN HÌNH ================= */
    let currentScreen = 1;
    let letterState = 0;

    function goToScreen(num) {
      document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
      const target = document.getElementById(`screen-${num}`);
      if (target) target.classList.add('active');
      currentScreen = num;

      if (num === 6) {
        document.body.classList.add('night-mode');
        renderNightGarden();
      } else {
        document.body.classList.remove('night-mode');
      }

      if (num === 2) runLoadingScreen();
      if (num === 3) runChibiAnimation();
      if (num === 5) {
        letterState = 0;
        document.getElementById('letter-modal').classList.remove('open');
        document.getElementById('letter-hint').innerText = 'CHẠM VÀO LÁ THƯ ĐỂ ĐỌC';
      }
    }

    function startExperience() {
      goToScreen(2);
    }

    function restartExperience() {
      goToScreen(1);
    }

    /* ================= 2. MÀN HÌNH 2: THANH LOADING ================= */
    function runLoadingScreen() {
      const fill = document.getElementById('progress-fill');
      let progress = 0;
      fill.style.width = '0%';
      const interval = setInterval(() => {
        progress += 2;
        fill.style.width = `${progress}%`;
        if (progress >= 100) {
          clearInterval(interval);
          setTimeout(() => goToScreen(3), 400);
        }
      }, 55);
    }

    /* ================= 3. MÀN HÌNH 3: HOẠT CẢNH VẼ CHIBI ================= */
    function runChibiAnimation() {
      for (let i = 1; i <= 5; i++) {
        document.getElementById(`part-${i}`).classList.remove('show');
      }
      document.getElementById('chibi-speech').classList.remove('show');

      const delays = [200, 1400, 2600, 3800, 5000];
      delays.forEach((delay, idx) => {
        setTimeout(() => {
          if (currentScreen === 3) {
            document.getElementById(`part-${idx + 1}`).classList.add('show');
          }
        }, delay);
      });

      setTimeout(() => {
        if (currentScreen === 3) {
          document.getElementById('chibi-speech').classList.add('show');
        }
      }, 5600);
    }

    /* ================= 4. MÀN HÌNH 5: ĐÓNG / MỞ THƯ ================= */
    function toggleLetter() {
      const modal = document.getElementById('letter-modal');
      const hint = document.getElementById('letter-hint');

      if (letterState === 0) {
        modal.classList.add('open');
        hint.innerText = 'CHẠM ĐỂ ĐÓNG THƯ';
        letterState = 1;
      } else if (letterState === 1) {
        modal.classList.remove('open');
        hint.innerText = 'CHẠM ĐỂ TIẾP TỤC ➔';
        letterState = 2;
      } else {
        goToScreen(6);
      }
    }

    function handleLetterHintClick() {
      toggleLetter();
    }

    /* ================= 5. MÀN HÌNH 6: VƯỜN HOA ĐÊM SAO ================= */
    function renderNightGarden() {
      const container = document.getElementById('garden-container');
      container.innerHTML = '';

      const flowersData = [
        { left: '22%', height: 280, scale: 0.85, delay: '0s' },
        { left: '38%', height: 390, scale: 1.05, delay: '0.3s' },
        { left: '53%', height: 430, scale: 1.15, delay: '0.6s' },
        { left: '72%', height: 330, scale: 0.95, delay: '0.2s' }
      ];

      flowersData.forEach(f => {
        const svgNS = "http://www.w3.org/2000/svg";
        const svg = document.createElementNS(svgNS, "svg");
        svg.setAttribute("class", "night-flower");
        svg.style.left = f.left;
        svg.style.width = "140px";
        svg.style.height = `${f.height}px`;
        svg.style.animationDelay = f.delay;
        svg.setAttribute("viewBox", "0 0 140 400");

        svg.innerHTML = `
          <path class="stem-grow" d="M 70 400 Q 55 250 70 80" fill="none" stroke="#3ca64b" stroke-width="5" stroke-linecap="round"/>
          <path class="bloom-pop" d="M 65 260 Q 25 240 35 280 Q 50 290 65 260 Z" fill="#4ec95e"/>
          <path class="bloom-pop" d="M 68 190 Q 110 170 100 210 Q 85 220 68 190 Z" fill="#4ec95e"/>
          <g class="bloom-pop" style="transform-origin: 70px 80px;">
            <circle cx="70" cy="48" r="22" fill="#ff9ec6" opacity="0.95"/>
            <circle cx="100" cy="68" r="22" fill="#ffb3d1" opacity="0.95"/>
            <circle cx="88" cy="102" r="22" fill="#ff85b3" opacity="0.95"/>
            <circle cx="52" cy="102" r="22" fill="#ff9ec6" opacity="0.95"/>
            <circle cx="40" cy="68" r="22" fill="#ffb3d1" opacity="0.95"/>
            <circle cx="70" cy="78" r="16" fill="#ffe066"/>
          </g>
        `;
        container.appendChild(svg);
      });

      setTimeout(() => {
        if (currentScreen === 6) goToScreen(7);
      }, 9500);
    }

    /* ================= 6. CANVAS HOA RƠI & TRÁI TIM BAY ================= */
    const canvas = document.getElementById('falling-canvas');
    const ctx = canvas.getContext('2d');
    let cw = canvas.width = window.innerWidth;
    let ch = canvas.height = window.innerHeight;

    window.addEventListener('resize', () => {
      cw = canvas.width = window.innerWidth;
      ch = canvas.height = window.innerHeight;
    });

    const icons = ['🌸', '🌺', '🌷', '💖', '✨'];
    const particles = Array.from({ length: 28 }, () => ({
      x: Math.random() * cw,
      y: Math.random() * ch,
      size: 14 + Math.random() * 12,
      speedY: 0.8 + Math.random() * 1.4,
      speedX: (Math.random() - 0.5) * 1.2,
      angle: Math.random() * 360,
      spin: (Math.random() - 0.5) * 2,
      icon: icons[Math.floor(Math.random() * icons.length)]
    }));

    function drawParticles() {
      ctx.clearRect(0, 0, cw, ch);

      particles.forEach(p => {
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate((p.angle * Math.PI) / 180);
        ctx.font = `${p.size}px serif`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(p.icon, 0, 0);
        ctx.restore();

        if (currentScreen === 6) {
          p.y -= p.speedY * 0.7;
          if (p.y < -30) p.y = ch + 30;
        } else {
          p.y += p.speedY;
          if (p.y > ch + 30) p.y = -30;
        }

        p.x += p.speedX;
        p.angle += p.spin;
      });

      requestAnimationFrame(drawParticles);
    }

    drawParticles();
